2.3 Infer the oldest cohort size

Some rows report data for 5-year cohorts (e.g., “25 - 29 Years”), whereas others contain cumulative counts (e.g., “70 Years & Over”). We’ll use the cumulative values to estimate the size of the oldest five-year cohort.

# Convert NA strings to actual NA values, convert to numeric, and add sex column
data_relevant_years <- data_relevant_years %>%
  mutate(across(-`Data Series`, ~ na_if(tolower(.), "na"))) %>%
  mutate(across(-`Data Series`, as.numeric)) %>%
  mutate(sex = rep(c("Male", "Female"), each = 25))
# Define replacement rules for cumulative age groups
replacement_rules <- tibble(
  target = c("65 - 69 Years", "70 - 74 Years", "75 - 79 Years", "80 - 84 Years", "85 - 89 Years"),
  source = c("65 Years & Over", "70 Years & Over", "75 Years & Over", "80 Years & Over", "85 Years & Over")
)

# Replace cumulative values with cohort-specific values
data_relevant_years <- data_relevant_years %>%
  left_join(replacement_rules, by = c("Data Series" = "target")) %>%
  left_join(
    data_relevant_years %>%
      select(`Data Series`, sex, all_of(years)) %>%
      rename(source = `Data Series`) %>%
      rename_with(~ paste0("src_", .x), all_of(years)),
    by = c("source" = "source", "sex" = "sex")
  ) %>%
  mutate(across(all_of(years), ~ coalesce(.x, get(paste0("src_", cur_column()))))) %>%
  select(-starts_with("src_"), -source)

# Check the results for the oldest cohorts
kable(
  filter(data_relevant_years, `Data Series` %in% replacement_rules$target) %>%
    select(`Data Series`, sex, all_of(years)) %>%
    head(),
  caption = "Age Cohorts After Replacing Cumulative Values"
)

2.4 Format Data

We’ll construct a data frame with these columns: - age: Lower age limit of the cohort - sex - year: Year of the data - pop: Population size

# Drop total population rows
data_relevant_years <- data_relevant_years[-c(1, 26), ]

# Remove cumulative rows like "70 Years & Over"
data_relevant_years <- data_relevant_years %>%
  filter(!str_detect(`Data Series`, "Years & Over"))

# Extract age lower bound
data_relevant_years <- data_relevant_years %>%
  mutate(age = as.numeric(str_extract(`Data Series`, "^\\d+")))

# Reshape from wide to long format
data_relevant_years <- data_relevant_years %>%
  select(age, sex, all_of(years)) %>%
  pivot_longer(
    cols = all_of(years),
    names_to = "year",
    values_to = "pop"
  ) %>%
  mutate(
    year = as.integer(year),
    sex = factor(sex, levels = c("Male", "Female"))
  ) %>%
  drop_na(pop) %>%
  arrange(age, sex, year)

# Filter to age <= 85
data_relevant_years <- data_relevant_years %>%
  filter(age <= 85)
